## Usage

```tsx
import { Stack } from "@/components/pdf/stack";
import { Text } from "@/components/pdf/text";
```

```tsx
<Stack gap="md">
  <Text>One</Text>
  <Text>Two</Text>
</Stack>
```

## API Reference

### Stack

| Prop        | Type             | Default    |
| ----------- | ---------------- | ---------- |
| `gap`       | `StackGap`       | 'md'       |
| `direction` | `StackDirection` | 'vertical' |
| `align`     | `StackAlign`     | 'start'    |
| `justify`   | `StackJustify`   | 'start'    |
| `wrap`      | `boolean`        | false      |
| `noWrap`    | `boolean`        | false      |
